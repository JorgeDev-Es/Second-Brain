# 1480. Running Sum of 1d Array

### Rust

pub fn running_sum(nums: Vec<i32>) -> Vec<i32> {
	let mut result: Vec<i32> = Vec::new();
	let mut actual_sum: i32 = 0;

	result.push(nums[0]);
	actual_sum = nums[0];
	for i in 1..=nums.len() - 1 {
	let num = nums[i];
	actual_sum += num;
	result.push(actual_sum);
	}
	
	return result;
}
