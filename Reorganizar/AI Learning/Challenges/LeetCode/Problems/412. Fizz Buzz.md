```Rust
pub fn fizz_buzz(n: i32) -> Vec<String> {  
    let mut result: Vec<String> = vec![];  
  
    for i in 1..n + 1 {  
        if i % 3 == 0 && i % 5 == 0 {  
            result.push("Fizz Buzz".to_string());  
        } else if i % 3 == 0 {  
            result.push("Fizz".to_string());  
        } else if i % 5 == 0 {  
            result.push("Buzz".to_string());  
        } else {  
            result.push(i.to_string());  
        }  
    }  
  
    return result;  
}
```
