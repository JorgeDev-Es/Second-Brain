### Pseudocódigo
```
class ListNode {
	value: int
	next?: ListNode
	constructor(value?: number, next?: ListNode)
}

addTwoNumbers(l1?: ListNode, l2?: ListNode): ListNode?
	dummyList: ListNode;
	currentList: ListNode = dummyList;
	
	carry: int = 0
	while l1 != null || l2 != null || carry != 0
		sum: int
		if l1 != null
			sum += l1.val
			l1 = l1.next
		
		if l2 != null
			sum += l2.val
			l2 = l2.next
		
		sum += carry
		carry = sum / 10
		ListNode *newList = new ListNode(sum % 10)
		currentList.next = newList
		currentList = currentList.next
	
	return dummyList.next
```
